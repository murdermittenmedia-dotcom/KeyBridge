cmake_minimum_required(VERSION 3.22)
project(KeyBridge VERSION 0.1.1)

if (MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_subdirectory(JUCE)

juce_add_plugin(KeyBridge
    COMPANY_NAME "KeyBridge"
    PLUGIN_MANUFACTURER_CODE KBrg
    PLUGIN_CODE KBr1
    FORMATS VST3
    PRODUCT_NAME "KeyBridge"
    DESCRIPTION "Transparent key, BPM, and Vocal Fit analysis"
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    VST3_CATEGORIES Fx
    VST3_AUTO_MANIFEST TRUE
    COPY_PLUGIN_AFTER_BUILD FALSE
)

juce_generate_juce_header(KeyBridge)

target_sources(KeyBridge PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
)

target_compile_definitions(KeyBridge PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(KeyBridge PRIVATE
    juce::juce_audio_utils
    juce::juce_dsp
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

if (MSVC)
    target_compile_options(KeyBridge PRIVATE /W4 /permissive- /EHsc)
    set_property(TARGET KeyBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
