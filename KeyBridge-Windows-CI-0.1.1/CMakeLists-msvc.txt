cmake_minimum_required(VERSION 3.22)
project(Tunerite VERSION 0.1.1)

if (MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_subdirectory(JUCE)

# Supplied Murder Mitten Media brand asset, embedded in the native editor.
juce_add_binary_data(TuneriteBrandingAssets
    SOURCES
        Resources/MurderMittenMediaLogo.png
)

juce_add_plugin(Tunerite
    COMPANY_NAME "Murder Mitten Media"
    PLUGIN_MANUFACTURER_CODE Mmm1
    PLUGIN_CODE Tnr1
    FORMATS VST3
    PRODUCT_NAME "TuneRite"
    DESCRIPTION "TuneRite by Murder Mitten Media — transparent beat key and BPM analyzer"
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    VST3_CATEGORIES Fx
    VST3_AUTO_MANIFEST TRUE
    COPY_PLUGIN_AFTER_BUILD FALSE
)

juce_generate_juce_header(Tunerite)

target_sources(Tunerite PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/AnalysisCore.cpp
    Source/VocalFit.cpp
)

target_compile_definitions(Tunerite PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(Tunerite PRIVATE
    TuneriteBrandingAssets
    juce::juce_audio_utils
    juce::juce_dsp
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

if (MSVC)
    target_compile_options(Tunerite PRIVATE /W4 /permissive- /EHsc)
    set_property(TARGET Tunerite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

juce_add_console_app(TuneriteOfflineAnalyzer
    PRODUCT_NAME "Tunerite Offline Analyzer"
)

juce_generate_juce_header(TuneriteOfflineAnalyzer)

target_sources(TuneriteOfflineAnalyzer PRIVATE
    Source/OfflineAnalyzer.cpp
    Source/AnalysisCore.cpp
)

target_link_libraries(TuneriteOfflineAnalyzer PRIVATE
    juce::juce_audio_formats
    juce::juce_audio_basics
    juce::juce_dsp
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

if (MSVC)
    target_compile_options(TuneriteOfflineAnalyzer PRIVATE /W4 /permissive- /EHsc)
    set_property(TARGET TuneriteOfflineAnalyzer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_executable(TuneriteAnalysisCoreSyntheticTests
    Source/AnalysisCoreSyntheticTests.cpp
    Source/AnalysisCore.cpp
)

target_compile_features(TuneriteAnalysisCoreSyntheticTests PRIVATE cxx_std_17)

target_include_directories(TuneriteAnalysisCoreSyntheticTests PRIVATE Source)
target_link_libraries(TuneriteAnalysisCoreSyntheticTests PRIVATE
    juce::juce_dsp
    juce::juce_audio_basics
)

if (MSVC)
    target_compile_options(TuneriteAnalysisCoreSyntheticTests PRIVATE /W4 /permissive- /EHsc)
    set_property(TARGET TuneriteAnalysisCoreSyntheticTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

juce_add_console_app(TuneriteProcessorCaptureTests
    PRODUCT_NAME "Tunerite Processor Capture Tests"
)

juce_generate_juce_header(TuneriteProcessorCaptureTests)

target_sources(TuneriteProcessorCaptureTests PRIVATE
    Source/PluginProcessorCaptureTests.cpp
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/AnalysisCore.cpp
    Source/VocalFit.cpp
)

target_link_libraries(TuneriteProcessorCaptureTests PRIVATE
    TuneriteBrandingAssets
    juce::juce_audio_utils
    juce::juce_dsp
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

if (MSVC)
    target_compile_options(TuneriteProcessorCaptureTests PRIVATE /W4 /permissive- /EHsc)
    set_property(TARGET TuneriteProcessorCaptureTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_executable(TuneriteVocalFitTests
    Source/VocalFitTests.cpp
    Source/VocalFit.cpp
)
target_compile_features(TuneriteVocalFitTests PRIVATE cxx_std_17)
target_include_directories(TuneriteVocalFitTests PRIVATE Source)
if (MSVC)
    target_compile_options(TuneriteVocalFitTests PRIVATE /W4 /permissive- /EHsc)
    set_property(TARGET TuneriteVocalFitTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

enable_testing()
add_test(NAME TuneriteAnalysisCoreSyntheticTests COMMAND TuneriteAnalysisCoreSyntheticTests)
add_test(NAME TuneriteProcessorCaptureTests COMMAND TuneriteProcessorCaptureTests)
add_test(NAME TuneriteVocalFitTests COMMAND TuneriteVocalFitTests)
